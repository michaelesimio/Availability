const DAYS = ['saturday','sunday','monday','tuesday','wednesday','thursday','friday'];

const DAY_LABEL = { saturday:'Saturday', sunday:'Sunday', monday:'Monday', tuesday:'Tuesday', wednesday:'Wednesday', thursday:'Thursday', friday:'Friday' };

const DAY_ABBR  = { saturday:'SAT', sunday:'SUN', monday:'MON', tuesday:'TUE', wednesday:'WED', thursday:'THU', friday:'FRI' };


// ================================================================

//  Week helpers

// ================================================================

function getSaturday(offsetWeeks = 0) {

  const d = new Date();

  d.setHours(0,0,0,0);

  const daysSinceSat = (d.getDay() + 1) % 7; // Sun=1, Mon=2,…Sat=0

  d.setDate(d.getDate() - daysSinceSat + offsetWeeks * 7);

  return d;

}


function toKey(d) { return d.toISOString().slice(0,10); }


function fmtRange(sat) {

  const fri = new Date(sat); fri.setDate(sat.getDate() + 6);

  const o = { month:'short', day:'numeric' };

  return sat.toLocaleDateString('en-US',o) + ' – ' + fri.toLocaleDateString('en-US',{...o,year:'numeric'});

}


function fmtShort(sat) {

  const fri = new Date(sat); fri.setDate(sat.getDate() + 6);

  const o = { month:'short', day:'numeric' };

  return sat.toLocaleDateString('en-US',o) + ' – ' + fri.toLocaleDateString('en-US',o);

}


// Default: next week, unless today is Sat or Sun → this week

let selWeek = (new Date().getDay() === 6 || new Date().getDay() === 0) ? 0 : 1;

const weekStarts = [getSaturday(0), getSaturday(1)];


// ================================================================

//  Build week selector

// ================================================================

function buildWeekGrid() {

  const el = document.getElementById('weekGrid');

  el.innerHTML = '';

  ['This Week','Next Week'].forEach((lbl, i) => {

    const wrap = document.createElement('div');

    wrap.className = 'week-opt';

    wrap.innerHTML = `

      <input type="radio" name="wk" id="wk${i}" value="${i}" ${i===selWeek?'checked':''} onchange="onWeekChange(${i})">

      <label for="wk${i}">

        <span class="wk-tag">${lbl}</span>

        <span class="wk-range">${fmtShort(weekStarts[i])}</span>

      </label>`;

    el.appendChild(wrap);

  });

}


function onWeekChange(idx) { selWeek = idx; buildDays(); }

function buildDays() {

  const el = document.getElementById('daysList');

function clickHeader(day) {

  const cb = document.getElementById('tog-' + day);

  cb.checked = !cb.checked;

  onToggle(day);

}


function onToggle(day) {

  const cb   = document.getElementById('tog-' + day);

  const card = document.getElementById('card-' + day);

  const lbl  = document.getElementById('tlbl-' + day);

  if (cb.checked) {

    card.classList.add('on');

    lbl.textContent = 'ON';

    if (!document.getElementById('blist-' + day).children.length) addBlock(day);

  } else {

    card.classList.remove('on');

    lbl.textContent = 'OFF';

  }

  updateProgress();

}


function addBlock(day) {

  const id  = 'blk-' + day + '-' + (blkId++);

  const div = document.createElement('div');

  div.className = 'time-block';

  div.id = id;

  div.innerHTML = `

    <input type="time" id="${id}-s" value="08:00" onchange="validateBlk('${id}')">

    <span class="sep-lbl">to</span>

    <input type="time" id="${id}-e" value="16:00" onchange="validateBlk('${id}')">

    <button class="btn-rm" onclick="rmBlock('${id}','${day}')" title="Remove">×</button>`;

  document.getElementById('blist-' + day).appendChild(div);

}


function rmBlock(id, day) {

  const el = document.getElementById(id);

  if (el) el.remove();

  // keep at least one block if day is toggled on

  const cb = document.getElementById('tog-' + day);

  if (cb && cb.checked && !document.getElementById('blist-' + day).children.length) addBlock(day);

}


function validateBlk(id) {

  const s = document.getElementById(id + '-s');

  const e = document.getElementById(id + '-e');

  if (s && e && s.value && e.value) {

    e.classList.toggle('error', s.value >= e.value);

  }

}


// ================================================================

//  Progress bar

// ================================================================

function onInfoChange() { updateProgress(); }


function updateProgress() {

  let pts = 0;

  if (document.getElementById('empName').value.trim()) pts++;

  if (document.getElementById('empDept').value) pts++;

  const activeDays = DAYS.filter(d => {

    const cb = document.getElementById('tog-' + d);

    return cb && cb.checked;

  }).length;

  pts += Math.min(activeDays, 5);

  document.getElementById('progressFill').style.width = Math.min((pts / 7) * 100, 100) + '%';

}


// ================================================================

//  Collect & validate

// ================================================================

function collect() {

  const name = document.getElementById('empName').value.trim();

  const dept = document.getElementById('empDept').value;

  const weekOf = toKey(weekStarts[selWeek]);

  const days = {};

  let valid = true;

  let errMsg = '';


  DAYS.forEach(day => {

    const cb = document.getElementById('tog-' + day);

    if (!cb || !cb.checked) { days[day] = []; return; }


    const blocks = [];

    const blistEl = document.getElementById('blist-' + day);

    for (const blk of blistEl.children) {

      const sEl = blk.querySelector('input[type="time"]:first-of-type');

      const eEl = blk.querySelector('input[type="time"]:last-of-type');

      if (!sEl || !eEl || !sEl.value || !eEl.value) {

        valid = false; errMsg = 'All time blocks need a start and end time.'; return;

      }

      if (sEl.value >= eEl.value) {

        valid = false; errMsg = 'End time must be after start time for all blocks.'; return;

      }

      blocks.push({ start: sEl.value, end: eEl.value });

    }

    days[day] = blocks;

  });


  return { name, dept, weekOf, days, valid, errMsg };

}

  const alertBox = document.getElementById('alertBox');

  const btn      = document.getElementById('submitBtn');

  const btnLbl   = document.getElementById('btnLabel');

  alertBox.className = 'alert';


  const { name, dept, weekOf, days, valid, errMsg } = collect();


  if (!name) return flash('err', '⚠ Please enter your full name.');

  if (!dept)  return flash('err', '⚠ Please select your department.');

  if (!valid) return flash('err', '⚠ ' + errMsg);

  if (!Object.values(days).some(d => d.length > 0))

    return flash('err', '⚠ Please mark at least one day as available and add time blocks.');

  if (APPS_SCRIPT_URL === 'https://script.google.com/macros/s/AKfycbyLVju589W3_7O-KNoBWe58IWc2TuRhi-YCY7IzIgK8BazC027wzOqOuj99r15h6FWg/exec')

    return flash('err', '⚙ This form hasn\'t been configured yet. Please contact your manager.');


  btn.disabled = true;

  btnLbl.textContent = 'Submitting…';


  const payload = { name, department: dept, weekOf, days };

  const url = APPS_SCRIPT_URL + '?action=submit&data=' + encodeURIComponent(JSON.stringify(payload));


  try {

    // Try with CORS first; fall back to no-cors if needed

    let ok = false;

    try {

      const res  = await fetch(url);

      const json = await res.json();

      ok = json.success;

    } catch (_) {

      // CORS blocked — fire-and-forget; data still reaches the sheet

      await fetch(url, { mode: 'no-cors' });

      ok = true; // optimistic

    }


    if (ok) {

      flash('ok', '✅ Availability submitted for the week of ' + fmtRange(weekStarts[selWeek]) + '. Thank you, ' + name + '!');

      setTimeout(() => {

        document.getElementById('empName').value = '';

        document.getElementById('empDept').value = '';

        buildDays();

        updateProgress();

        alertBox.className = 'alert';

      }, 4000);

    } else {

      flash('err', '❌ Submission was not saved. Please try again.');

    }

  } catch (err) {

    flash('err', '❌ Network error — check your connection and try again. (' + err.message + ')');

  } finally {

    btn.disabled = false;

    btnLbl.textContent = 'Submit Availability';

  }

}


function flash(type, msg) {

  const el = document.getElementById('alertBox');

  el.className = 'alert show ' + type;

  el.textContent = msg;

}
