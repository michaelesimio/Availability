// ================================================================

//  Week helpers

// ================================================================

function getSaturday(offsetWeeks = 0) {

  const d = new Date();

  d.setHours(0,0,0,0);

  const daysSinceSat = (d.getDay() + 1) % 7;

  d.setDate(d.getDate() - daysSinceSat + offsetWeeks * 7);

  return d;

}

function toKey(d) { return d.toISOString().slice(0,10); }

function fmtRange(sat) {

  const fri = new Date(sat); fri.setDate(sat.getDate() + 6);

  const o = { month:'short', day:'numeric' };

  return sat.toLocaleDateString('en-US',o) + ' – ' + fri.toLocaleDateString('en-US',{...o,year:'numeric'});

}


// ================================================================

//  Build week dropdown (prev week through 3 weeks out)

// ================================================================

(function buildWeekSel() {

  const sel = document.getElementById('weekSel');

  for (let i = -1; i <= 3; i++) {

    const sat = getSaturday(i);

    const opt = document.createElement('option');

    opt.value       = toKey(sat);

    opt.textContent = fmtRange(sat);

    if (i === 0) opt.selected = true;

    sel.appendChild(opt);

  }

})();

      renderTable();

    } else {

      showErr('❌ Server error: ' + (json.error || 'Unknown error.'));

      setTableMsg('Failed to load. See error above.', '⚠');

    }

  } catch (err) {

    showErr('❌ Could not reach the Apps Script. Make sure your URL is correct and the script is deployed as a Web App with "Anyone" access. Error: ' + err.message);

    setTableMsg('Connection failed. See error above.', '⚠');

  } finally {

    btn.disabled = false; btn.textContent = 'Load';

  }

}

function renderTable() {

  const data = activeFilter === 'all'

    ? submissions

    : submissions.filter(s => s.department === activeFilter);


  if (!data.length) {

    setTableMsg(

      submissions.length === 0

        ? 'No submissions found for this week yet.'

        : 'No ' + activeFilter + ' submissions for this week.',

      '📭'

    );

    return;

  }


  const dayHeaders = DAYS.map(d => `<th>${DAY_SHORT[d]}</th>`).join('');


  const rows = data.map(s => {

    const dc = s.department === 'Passenger Service' ? 'ps' : 'ramp';

    const da = s.department === 'Passenger Service' ? 'PS'  : 'Ramp';


    const dayCells = DAYS.map(day => {

      const blocks = (s.days && s.days[day]) || [];

      if (!blocks.length) return `<td><span class="unavail">—</span></td>`;

      const chips = blocks.map(b =>

        `<span class="chip">${fmt12(b.start)}–${fmt12(b.end)}</span>`

      ).join('');

      return `<td><div class="chips">${chips}</div></td>`;

    }).join('');


    const ts = s.timestamp

      ? new Date(s.timestamp).toLocaleDateString('en-US',{month:'short',day:'numeric',hour:'numeric',minute:'2-digit'})

      : '—';


    return `<tr>

      <td><span class="emp-name">${esc(s.name)}</span></td>

      <td><span class="dept-badge ${dc}">${da}</span></td>

      ${dayCells}

      <td style="color:var(--text3);font-size:11px;white-space:nowrap">${ts}</td>

    </tr>`;

  }).join('');


  document.getElementById('tblContainer').innerHTML = `

    <div class="tbl-wrap">

      <div class="tbl-scroll">

        <table>

          <thead><tr>

            <th>Name</th><th>Dept</th>${dayHeaders}<th>Submitted</th>

          </tr></thead>

          <tbody>${rows}</tbody>

        </table>

      </div>

    </div>`;

}

function findAvailable() {

  const dept       = document.getElementById('s-dept').value;

  const day        = document.getElementById('s-day').value;

  const shiftStart = document.getElementById('s-start').value;

  const shiftEnd   = document.getElementById('s-end').value;


  if (!dept || !day || !shiftStart || !shiftEnd) {

    return alert('Please fill in all four shift fields before searching.');

  }

  if (shiftStart >= shiftEnd) {

    return alert('Shift end time must be after shift start time.');

  }

  if (!submissions.length) {

    return showSchedMsg('📭',

      'No availability data is loaded.<br>Go to the <strong>Availability Overview</strong> tab, pick a week, and click <strong>Load</strong> first.'

    );

  }


  // Filter to this department, then check each employee's blocks for that day

  const deptEmps = submissions.filter(s => s.department === dept);

  const available = [];


  deptEmps.forEach(emp => {

    const blocks = (emp.days && emp.days[day]) || [];

    // An employee CAN work the shift if any block fully contains it

    const matching = blocks.filter(b => b.start <= shiftStart && b.end >= shiftEnd);

    if (matching.length) available.push({ name: emp.name, dept: emp.department, matching });

  });


  renderResults(available, dept, day, shiftStart, shiftEnd, deptEmps.length);

}


function renderResults(avail, dept, day, start, end, totalChecked) {

  const ph  = document.getElementById('resPlaceholder');

  const box = document.getElementById('resContent');

  ph.style.display  = 'none';

  box.style.display = 'block';


  const countClass = avail.length === 0 ? 'zero' : '';

  const shiftLabel = fmt12(start) + ' – ' + fmt12(end);

  const dayLabel   = DAY_LABEL[day];

  const weekLabel  = document.getElementById('weekSel').options[document.getElementById('weekSel').selectedIndex]?.text || '';


  let bodyHtml;

  if (avail.length === 0) {

    bodyHtml = `

      <div style="text-align:center;padding:36px 16px;color:var(--text3);">

        <div style="font-size:34px;margin-bottom:12px">😔</div>

        <p style="font-size:14px;line-height:1.7">

          No ${dept} employees are available for this shift.<br>

          <span style="font-size:12px">(${totalChecked} ${dept} submission${totalChecked !== 1 ? 's' : ''} checked)</span>

        </p>

      </div>`;

  } else {

    const cards = avail.map((emp, idx) => {

      const dc = emp.dept === 'Passenger Service' ? 'ps' : 'ramp';

      const da = emp.dept === 'Passenger Service' ? 'Passenger Service' : 'Ramp';

      const matchChips = emp.matching.map(b =>

        `<span class="match-chip">✓ ${fmt12(b.start)} – ${fmt12(b.end)}</span>`

      ).join('');

      return `

        <div class="result-card" style="animation-delay:${idx * 40}ms">

          <div class="rc-name">${esc(emp.name)}</div>

          <div class="rc-dept"><span class="dept-badge ${dc}" style="font-size:10px">${da}</span></div>

          <div class="rc-blocks">${matchChips}</div>

        </div>`;

    }).join('');


    bodyHtml = `<div class="result-list">${cards}</div>`;

  }


  box.innerHTML = `

    <div class="results-header">

      <div class="count-bubble ${countClass}">${avail.length}</div>

      <div class="results-meta">

        <strong>${avail.length} employee${avail.length !== 1 ? 's' : ''} available</strong>

        <span>${dept} · ${dayLabel} · ${shiftLabel}</span>

        <span style="font-size:11px">${weekLabel}</span>

      </div>

    </div>

    ${bodyHtml}`;

}


function showSchedMsg(icon, html) {

  const ph  = document.getElementById('resPlaceholder');

  const box = document.getElementById('resContent');

  ph.style.display  = 'none';

  box.style.display = 'block';

  box.innerHTML     = `

    <div style="text-align:center;padding:48px 20px;color:var(--text3)">

      <div style="font-size:36px;margin-bottom:12px;opacity:.45">${icon}</div>

      <p style="font-size:14px;line-height:1.6">${html}</p>

    </div>`;

}

function fmt12(t) {

  if (!t) return '';

  const [hh, mm] = t.split(':');

  let h = parseInt(hh, 10);

  const ampm = h < 12 ? 'am' : 'pm';

  if (h === 0) h = 12; else if (h > 12) h -= 12;

  return `${h}:${mm}${ampm}`;

}


function esc(s) {

  return String(s)

    .replace(/&/g,'&amp;').replace(/</g,'&lt;')

    .replace(/>/g,'&gt;').replace(/"/g,'&quot;');

}

  const el = document.getElementById('errBanner');

  el.classList.add('show');

}
